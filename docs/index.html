5. Audit Pixel Perfect (was hero not centered)

9 Compute absolute positions

function getAbs(key){
  let x=0,y=0, cur=map.get(key);
  while(cur){ x+=cur.transform.m02||0; y+=cur.transform.m12||0;
    cur=map.get(cur.parentIndex.guid) }
  return {x,y}
}

10 Animations from Figma interactions

// 1. Reveal on scroll — IntersectionObserver (modern, efficient)
const reveals = document.querySelectorAll('.reveal');
const io = new IntersectionObserver(entries=>{
  entries.forEach(e=>{
    if(e.isIntersecting){ e.target.classList.add('is-visible'); io.unobserve(e.target)}
  })
},{threshold:0.12});
reveals.forEach(el=>io.observe(el));
// CSS: .reveal{opacity:0; transform:translateY(18px)} .is-visible{opacity:1}

// 2. Header hide on scroll down
let lastY=0;
window.addEventListener('scroll',()=>{
  const y=window.scrollY;
  if(y>400 && y>lastY) header.style.transform='translateX(-50%) translateY(-88px)';
  else header.style.transform='translateX(-50%) translateY(0)';
  lastY=y;
},{passive:true});

// 3. Smooth scroll with header offset
document.querySelectorAll('a[href^="#"]').forEach(a=>{
  a.addEventListener('click',e=>{
    const target=document.querySelector(a.getAttribute('href'));
    if(target){ e.preventDefault();
      window.scrollTo({top:target.getBoundingClientRect().top+scrollY-72, behavior:'smooth'});
    }
  });
});

// 4. Card 3D tilt on mousemove
card.addEventListener('mousemove',e=>{
  const r=card.getBoundingClientRect();
  const x=(e.clientX-r.left)/r.width-0.5;
  const y=(e.clientY-r.top)/r.height-0.5;
  card.style.transform=`translateY(-4px) perspective(800px) rotateX(${-y*4}deg) rotateY(${x*6}deg)`;
});

All functions use basic JS: querySelector, addEventListener, classList, style.transform.
